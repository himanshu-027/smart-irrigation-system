#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

int sensorPin = A0;
int relayPin = 8;
int moisture;

void setup() {
  Serial.begin(9600);

  pinMode(relayPin, OUTPUT);
  digitalWrite(relayPin, HIGH);

  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print("Smart Irrigation");

  delay(1500);
  lcd.clear();
}

void loop() {
  moisture = analogRead(sensorPin);

  Serial.print("Moisture Level: ");
  Serial.println(moisture);

  lcd.setCursor(0, 0);
  lcd.print("Moisture: ");
  lcd.print(moisture);
  lcd.print("   ");

  if (moisture > 700) {
    digitalWrite(relayPin, LOW);

    Serial.println("PUMP: ON");

    lcd.setCursor(0, 1);
    lcd.print("Pump: ON ");
  }
  else {
    digitalWrite(relayPin, HIGH);

    Serial.println("PUMP: OFF");

    lcd.setCursor(0, 1);
    lcd.print("Pump: OFF");
  }

  delay(1000);
}
